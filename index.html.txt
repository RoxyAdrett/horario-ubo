  <script>
    const days = ['lun', 'mar', 'mie', 'jue', 'vie'];
    function setDay(d) {
      document.querySelectorAll('.day-content').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.nav button').forEach(el => el.classList.remove('active'));
      document.getElementById(d).classList.add('active');
      const idx = days.indexOf(d);
      if (idx !== -1) document.querySelectorAll('.nav button')[idx].classList.add('active');
    }
    const today = new Date().getDay(); // 1=Lun, 5=Vie
    if (today >= 1 && today <= 5) setDay(days[today - 1]);
    else setDay('lun');
  </script>
